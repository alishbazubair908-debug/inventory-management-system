// SYSTEM DATE & TIME UTILITY
// ==========================================
inline std::string getCurrentDateTime() {
    std::time_t now = std::time(nullptr);
    std::tm ltm;
#if defined(_MSC_VER)
    localtime_s(&ltm, &now);
#else
    std::tm* ltm_ptr = std::localtime(&now);
    if (ltm_ptr) {
        ltm = *ltm_ptr;
    }
    else {
        std::memset(&ltm, 0, sizeof(std::tm));
    }
#endif
    char buf[64];
    std::strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", &ltm);
    return std::string(buf);
}

static std::string trim(const std::string& str) {
    if (str.empty()) return str;
    size_t first = str.find_first_not_of(" \t\r\n");
    if (first == std::string::npos) return "";
    size_t last = str.find_last_not_of(" \t\r\n");
    return str.substr(first, (last - first + 1));
}

// 19. BARCODE CLASS (Utility)
// ==========================================
class Barcode {
private:
    std::string barcodeValue;
    std::string format; // EAN13, QR, UPC

public:
    Barcode() : barcodeValue("000000000000"), format("UPC") {}

    Barcode(const std::string& val, const std::string& fmt) : format(fmt) {
        setValue(val);
    }

    void setValue(std::string val) {
        if (format == "EAN13" && (val.length() != 13 || val.find_first_not_of("0123456789") != std::string::npos)) {
            barcodeValue = "0000000000000";
        }
        else if (format == "UPC" && (val.length() != 12 || val.find_first_not_of("0123456789") != std::string::npos)) {
            barcodeValue = "000000000000";
        }
        else {
            barcodeValue = val;
        }
    }

    std::string getValue() const { return barcodeValue; }
    std::string getFormat() const { return format; }

    bool validate() const {
        if (format == "EAN13") return barcodeValue.length() == 13;
        if (format == "UPC") return barcodeValue.length() == 12;
        return !barcodeValue.empty();
    }

    std::string scan() const {
        return "[" + format + "] " + barcodeValue;
    }

    bool operator==(const Barcode& other) const {
        return (barcodeValue == other.barcodeValue && format == other.format);
    }

    friend std::ostream& operator<<(std::ostream& os, const Barcode& b) {
        os << b.scan();
        return os;
    }
};

// ==========================================
// 16. DISCOUNT POLICY (Abstract Strategy)
// ==========================================
class DiscountPolicy {
protected:
    std::string policyName;
    std::string startDate;
    std::string endDate;

public:
    DiscountPolicy(std::string name, std::string start, std::string end)
        : policyName(name), startDate(start), endDate(end) {
    }

    virtual ~DiscountPolicy() {}

    virtual double calculate(double price) = 0;

    virtual bool isActive() const;

    std::string getPolicyName() const { return policyName; }

    friend std::ostream& operator<<(std::ostream& os, const DiscountPolicy& dp) {
        os << "Policy: " << dp.policyName << " [" << dp.startDate << " to " << dp.endDate << "]";
        return os;
    }
};

// ==========================================
// 17. BULK DISCOUNT CLASS (Concrete Strategy)
// ==========================================
class BulkDiscount : public DiscountPolicy {
private:
    int minQuantity;
    double discountPercent;
    int currentQuantity;

public:
    BulkDiscount(std::string name, std::string start, std::string end, int minQty, double percent)
        : DiscountPolicy(name, start, end), minQuantity(minQty), discountPercent(percent), currentQuantity(0) {
    }

    void setCurrentQuantity(int qty) { currentQuantity = qty; }

    double calculate(double price) override {
        if (isActive()) {
            return price * (1.0 - (discountPercent / 100.0));
        }
        return price;
    }

    bool isActive() const override {
        return DiscountPolicy::isActive() && (currentQuantity >= minQuantity);
    }
};

// ==========================================
// 18. SEASONAL DISCOUNT CLASS (Concrete Strategy)
// ==========================================
class SeasonalDiscount : public DiscountPolicy {
private:
    std::string seasonName;
    double discountRate;

public:
    SeasonalDiscount(std::string name, std::string start, std::string end, std::string season, double rate)
        : DiscountPolicy(name, start, end), seasonName(season), discountRate(rate) {
    }

    double calculate(double price) override {
        if (isActive()) {
            return price * (1.0 - discountRate);
        }
        return price;
    }

    bool isActive() const override { return DiscountPolicy::isActive(); }

    std::string getSeasonName() const { return seasonName; }
};

// ==========================================
// 01. PRODUCT CLASS (Abstract Root)
// ==========================================
class Product {
protected:
    std::string productID;
    std::string productName;
    double price;
    double originalPrice;
    int quantity;
    Barcode barcode;
    std::string activeDiscountPolicyName;
    int storageNumber = 1;

public:
    Product(std::string id, std::string name, double p, int qty, Barcode b)
        : productID(id), productName(name), price(p), originalPrice(p), quantity(qty), barcode(b), activeDiscountPolicyName("None") {
    }

    virtual ~Product() {}

    virtual std::unique_ptr<Product> clone() const = 0;
    virtual double calculateValue() const { return price * quantity; };

    virtual void displayStatus() = 0;
    virtual double calculateRisk() = 0;
    virtual void applyDiscount(DiscountPolicy& policy) = 0;
    virtual std::string getCategory() const = 0;
    virtual std::string serialize() const = 0;

    virtual std::string getSubtype() const { return "Standard"; }
    virtual int getWarrantyMonths() const { return 0; }

    virtual void checkSafety() {
        std::cout << "[Safety Check] Product " << productName << " passes baseline standard audits." << std::endl;
    }

    void printDiscountInfo() const {
        if (activeDiscountPolicyName != "None" && price < originalPrice) {
            double pct = getDiscountPercent();
            std::cout << " [SALE: Save " << std::fixed << std::setprecision(0) << pct
                << "%! Original: $" << std::fixed << std::setprecision(2) << originalPrice
                << " | Current: $" << price << " (via " << activeDiscountPolicyName << ")]";
        }
    }

    double getDiscountPercent() const {
        if (originalPrice <= 0.0) return 0.0;
        return ((originalPrice - price) / originalPrice) * 100.0;
    }

    std::string getProductID() const { return productID; }
    std::string getProductName() const { return productName; }
    double getPrice() const { return price; }
    int getQuantity() const { return quantity; }
    Barcode getBarcode() const { return barcode; }

    void setPrice(double p) { price = p; }
    void setQuantity(int qty) { quantity = qty; }

    void setStorageNumber(int num) { storageNumber = num; }
    int getStorageNumber() const { return storageNumber; }

    bool operator==(const Product& other) const {
        return productID == other.productID;
    }

    friend std::ostream& operator<<(std::ostream& os, const Product& p) {
        os << "ID: " << p.productID << " | Name: " << p.productName
            << " | Price: $" << std::fixed << std::setprecision(2) << p.price
            << " | Qty: " << p.quantity << " | Barcode: " << p.barcode;
        if (p.activeDiscountPolicyName != "None") {
            os << " [Discounted from $" << p.originalPrice << " via " << p.activeDiscountPolicyName << "]";
        }
        return os;
    }
};

// 02. ELECTRONIC PRODUCT CLASS (Level 2)
// ==========================================
class ElectronicProduct : public Product {
protected:
    int voltage;
    int warrantyMonths;
    std::string brandName;

public:
    ElectronicProduct(std::string id, std::string name, double p, int qty, Barcode b,
        int volt, int warranty, std::string brand)
        : Product(id, name, p, qty, b), voltage(volt), warrantyMonths(warranty), brandName(brand) {
    }

    ~ElectronicProduct() override = default;

    std::unique_ptr<Product> clone() const override {
        return std::make_unique<ElectronicProduct>(*this);
    }

    bool testHardware() const { return (voltage >= 110 && voltage <= 240); }

    std::string getWarrantyStatus() const {
        std::ostringstream oss;
        oss << warrantyMonths << " Month Warranty from " << brandName;
        return oss.str();
    }

    std::string getCategory() const override { return "Electronic"; }
    int getWarrantyMonths() const override { return warrantyMonths; }

    std::string serialize() const override {
        std::ostringstream oss;
        oss << productID << "," << productName << "," << price << "," << quantity << ",Electronic,Standard," << storageNumber << ","
            << barcode.getValue() << "," << barcode.getFormat() << "," << voltage << "," << warrantyMonths << "," << brandName;
        return oss.str();
    }

    void displayStatus() override {
        std::cout << "[Electronic] " << productName << " (ID: " << productID
            << ") | Brand: " << brandName << " | Voltage: " << voltage
            << "V | Price: $" << std::fixed << std::setprecision(2) << price
            << " | Stock: " << quantity;
        printDiscountInfo();
        std::cout << std::endl;
    }

    double calculateRisk() override {
        return (static_cast<double>(voltage) * 0.02) + (36.0 / (static_cast<double>(warrantyMonths) + 1.0));
    }

    void applyDiscount(DiscountPolicy& policy) override {
        BulkDiscount* bulk = dynamic_cast<BulkDiscount*>(&policy);
        if (bulk) {
            bulk->setCurrentQuantity(quantity);
        }
        if (policy.isActive()) {
            activeDiscountPolicyName = policy.getPolicyName();
            setPrice(policy.calculate(originalPrice));
        }
        else {
            std::cout << "[Notice] Discount Policy '" << policy.getPolicyName() << "' is not currently active." << std::endl;
        }
    }
};

// ==========================================
// 08. STORAGE SECTION CLASS (Composition)
// ==========================================
class Storage {
private:
    int storageNumber;
    int capacity;
    std::string allowedCategory;
    std::vector<std::unique_ptr<Product>> products;

public:
    Storage() : storageNumber(0), capacity(10), allowedCategory("Any") {}

    Storage(int number, int cap, std::string restriction)
        : storageNumber(number), capacity(cap), allowedCategory(restriction) {
    }

    Storage(const Storage& other)
        : storageNumber(other.storageNumber), capacity(other.capacity), allowedCategory(other.allowedCategory) {
        for (const auto& item : other.products) {
            if (item) {
                products.push_back(item->clone());
            }
        }
    }

    Storage& operator=(const Storage& other) {
        if (this != &other) {
            storageNumber = other.storageNumber;
            capacity = other.capacity;
            allowedCategory = other.allowedCategory;
            products.clear();
            for (const auto& item : other.products) {
                if (item) {
                    products.push_back(item->clone());
                }
            }
        }
        return *this;
    }

    Storage(Storage&&) noexcept = default;
    Storage& operator=(Storage&&) noexcept = default;
    ~Storage() = default;

    int getStorageNumber() const { return storageNumber; }
    int getProductCount() const { return static_cast<int>(products.size()); }
    int getCapacity() const { return capacity; }
    std::string getAllowedCategory() const { return allowedCategory; }

    bool addStock(std::unique_ptr<Product> p) {
        if (allowedCategory != "Any" && p->getCategory() != allowedCategory) {
            std::cout << "\n[Error] Cannot place item in this area!" << std::endl;
            std::cout << "  -> " << p->getProductName() << " is of category: " << p->getCategory() << std::endl;
            std::cout << "  -> Storage " << storageNumber << " only accepts: " << allowedCategory << std::endl;
            return false;
        }
        p->setStorageNumber(storageNumber);
        products.push_back(std::move(p));
        return true;
    }

    std::unique_ptr<Product> takeProduct(int index) {
        if (index < 0 || index >= static_cast<int>(products.size())) return nullptr;
        std::unique_ptr<Product> p = std::move(products[index]);
        products.erase(products.begin() + index);
        return p;
    }

    void removeStock(const std::string& id) {
        products.erase(std::remove_if(products.begin(), products.end(),
            [&id](const std::unique_ptr<Product>& p) { return p->getProductID() == id; }),
            products.end());
    }

    void sortByID() {
        std::sort(products.begin(), products.end(),
            [](const std::unique_ptr<Product>& a, const std::unique_ptr<Product>& b) {
                return a->getProductID() < b->getProductID();
            });
    }

    double getTotalValue() const {
        double total = 0.0;
        for (const auto& p : products) {
            if (p) {
                total += p->calculateValue();
            }
        }
        return total;
    }

    void checkAllExpiries() {
        std::cout << "\n>>> Conducting Dynamic Safety and Expiry Audits inside Storage " << storageNumber << " <<<" << std::endl;
        bool checkedAny = false;
        for (auto& p : products) {
            if (p) {
                if (p->getSubtype() == "Perishable") {
                    std::cout << "  * [Perishable Check] ";
                    p->checkSafety();
                    checkedAny = true;
                }
                else if (p->getSubtype() == "Non-Perishable") {
                    std::cout << "  * [Preservation Check] ";
                    p->checkSafety();
                    checkedAny = true;
                }
            }
        }
        if (!checkedAny) {
            std::cout << "  -> No perishable or preservation-stabilized grocery profiles found in this storage." << std::endl;
        }
    }

    Product* operator[](int index) {
        if (index < 0 || index >= static_cast<int>(products.size())) {
            std::cerr << "Error: Storage pointer index out of bounds." << std::endl;
            return nullptr;
        }
        return products[index].get();
    }

    const Product* operator[](int index) const {
        if (index < 0 || index >= static_cast<int>(products.size())) {
            std::cerr << "Error: Storage pointer index out of bounds." << std::endl;
            return nullptr;
        }
        return products[index].get();
    }

    Storage operator+(const Storage& other) {
        Storage merged(storageNumber, capacity + other.capacity, allowedCategory);
        for (const auto& p : products) {
            if (p) merged.addStock(p->clone());
        }
        for (const auto& p : other.products) {
            if (p) merged.addStock(p->clone());
        }
        return merged;
    }

    friend std::ostream& operator<<(std::ostream& os, const Storage& s) {
        os << "Storage Section " << s.storageNumber << " [" << s.allowedCategory << " category only] (" << s.products.size() << "/" << s.capacity << " Occupied):\n";
        if (s.products.empty()) {
            os << "  (Storage currently empty - awaiting employee stock entry)\n";
        }
        else {
            for (size_t i = 0; i < s.products.size(); ++i) {
                os << "  [Item " << i + 1 << "] ";
                s.products[i]->displayStatus();
            }
        }
        return os;
    }
};

        if (index < 0 || index >= static_cast<int>(storages.size())) {
            std::cerr << "[FATAL] getStorage(): index " << index
                << " out of range (size=" << storages.size() << "). Aborting." << std::endl;
            std::abort();
        }
        return storages[index];
    }

    const Storage& getStorage(int index) const {
        if (index < 0 || index >= static_cast<int>(storages.size())) {
            std::cerr << "[FATAL] getStorage() const: index " << index
                << " out of range (size=" << storages.size() << "). Aborting." << std::endl;
            std::abort();
        }
        return storages[index];
    }

    void addStorage(Storage st) {
        storages.push_back(std::move(st));
    }

    double getGlobalValue() const {
        double total = 0.0;
        for (const auto& st : storages) {
            total += st.getTotalValue();
        }
        return total;
    }

    void removeStorageByIndex(int index) {
        if (index >= 0 && index < static_cast<int>(storages.size())) {
            storages.erase(storages.begin() + index);
        }
    }

    void findShortages() const {
        std::cout << "[Low Stock Alerts] Scanning shortages in '" << locationID << "':" << std::endl;
        bool lowFound = false;
        for (const auto& st : storages) {
            for (int j = 0; j < st.getProductCount(); ++j) {
                const Product* item = st[j];
                if (item && item->getQuantity() < 5) {
                    std::cout << "  - ALERT: Storage " << st.getStorageNumber()
                        << " | Item " << item->getProductName() << " (ID: "
                        << item->getProductID() << ") has quantity " << item->getQuantity() << std::endl;
                    lowFound = true;
                }
            }
        }
        if (!lowFound) {
            std::cout << "  -> Storage profiles are healthy." << std::endl;
        }
    }

    Product* findProductByID(const std::string& id) {
        for (auto& st : storages) {
            for (int j = 0; j < st.getProductCount(); ++j) {
                Product* p = st[j];
                if (p && p->getProductID() == id) {
                    return p;
                }
            }
        }
        return nullptr;
    }

    const Product* findProductByID(const std::string& id) const {
        for (const auto& st : storages) {
            for (int j = 0; j < st.getProductCount(); ++j) {
                const Product* p = st[j];
                if (p && p->getProductID() == id) {
                    return p;
                }
            }
        }
        return nullptr;
    }

    friend std::ostream& operator<<(std::ostream& os, const Warehouse& wh) {
        os << "==========================================\n"
            << "Warehouse: " << wh.locationID << " (" << wh.totalSquareFootage << " Sq.Ft.)\n"
            << "Active Storage Sections: " << wh.storages.size() << "\n"
            << "==========================================\n";
        for (const auto& st : wh.storages) {

template<typename T>
class TransactionLog {

    std::vector<T> logHistory;
    std::vector<std::string> timestamps;

public:
    TransactionLog() : logID("LOG-DEFAULT") {}
    TransactionLog(std::string id) : logID(id) {}

    void recordAction(T item) {
        logHistory.push_back(item);
        timestamps.push_back(getCurrentDateTime());
    }

    void recordActionWithTimestamp(T item, std::string ts) {
        logHistory.push_back(item);
        timestamps.push_back(ts);
    }

    int getLogCount() const { return static_cast<int>(logHistory.size()); }
    T getLogEntry(int idx) const { return logHistory[idx]; }
    std::string getTimestamp(int idx) const { return timestamps[idx]; }

    void printAuditTrail() const {
        std::cout << ">>> TIME-STAMPED AUDIT TRAIL FOR " << logID << " <<<" << std::endl;
        for (size_t i = 0; i < logHistory.size(); ++i) {
            std::cout << "  " << i + 1 << ". [" << timestamps[i] << "] " << logHistory[i] << std::endl;
        }
    }

    void printAuditTrailToFile(std::ofstream& file) const {
        file << ">>> TIME-STAMPED AUDIT TRAIL FOR " << logID << " <<<\n";
        for (size_t i = 0; i < logHistory.size(); ++i) {
            file << "  " << i + 1 << ". [" << timestamps[i] << "] " << logHistory[i] << "\n";
        }
    }

    T getEntry(int index) const {
        if (index < 0 || index >= static_cast<int>(logHistory.size())) {
            std::cerr << "Error: Out of bounds. Returning default instantiated type." << std::endl;
            return T();
        }
        return logHistory[index];
    }

    void clearLog() {
        logHistory.clear();
        timestamps.clear();
    }

    friend std::ostream& operator<<(std::ostream& os, const TransactionLog<T>& tl) {
        os << "Log [" << tl.logID << "] holds " << tl.logHistory.size() << " operations.";
        return os;
    }
};

static bool isAllAlphaOrSpace(const std::string& str) {
    if (str.empty()) return false;
    for (char c : str) {
        if (!((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || c == ' ')) {
            return false;
        }
    }
    return true;
}

static void flushInput() {
    std::cin.clear();
    while (std::cin.get() != '\n' && std::cin.good());
}

static std::string readString() {
    std::string val;
    std::getline(std::cin, val);
    return val;
}

static std::string readAlphaString() {
    std::string val;
    while (true) {
        std::getline(std::cin, val);
        if (isAllAlphaOrSpace(val)) {
            break;
        }
        std::cout << "Invalid. Use alphabetic characters and spaces only: ";
    }
    return val;
}

static bool isValidDate(const std::string& str) {
    if (str.length() != 10) return false;
    if (str[4] != '-' || str[7] != '-') return false;
    for (int i = 0; i < 10; ++i) {
        if (i == 4 || i == 7) continue;
        if (str[i] < '0' || str[i] > '9') return false;
    }
    int y = (str[0] - '0') * 1000 + (str[1] - '0') * 100 + (str[2] - '0') * 10 + (str[3] - '0');
    int m = (str[5] - '0') * 10 + (str[6] - '0');
    int d = (str[8] - '0') * 10 + (str[9] - '0');
    if (m < 1 || m > 12) return false;
    if (d < 1) return false;
    bool isLeap = (y % 4 == 0 && y % 100 != 0) || (y % 400 == 0);
    int maxDays = 31;
    if (m == 4 || m == 6 || m == 9 || m == 11) {
        maxDays = 30;
    }
    else if (m == 2) {
        maxDays = isLeap ? 29 : 28;
    }
    return d <= maxDays;
}

static bool isDateBeforeOrEqual(const std::string& date1, const std::string& date2) {
    return date1 <= date2;
}

static std::string readDate() {
    std::string input;
    while (true) {
        std::getline(std::cin, input);
        if (isValidDate(input)) {
            break;
        }
        std::cout << "Invalid date or calendar boundaries. Format as YYYY-MM-DD: ";
    }
    return input;
}

static int parseYear(const std::string& dateStr) {
    if (dateStr.length() < 4) return 0;
    int yr = 0;
    for (int i = 0; i < 4; ++i) {
        if (dateStr[i] >= '0' && dateStr[i] <= '9') {
            yr = yr * 10 + (dateStr[i] - '0');
        }
    }
    return yr;
}

static std::string addMonthsToDate(const std::string& dateStr, int months) {
    if (dateStr.length() != 10) return dateStr;
    int y = parseYear(dateStr);
    int m = (dateStr[5] - '0') * 10 + (dateStr[6] - '0');
    int d = (dateStr[8] - '0') * 10 + (dateStr[9] - '0');
    m += months;
    while (m > 12) {
        m -= 12;
        y += 1;
    }
    std::ostringstream oss;
    oss << y << "-" << std::setw(2) << std::setfill('0') << m << "-" << std::setw(2) << std::setfill('0') << d;
    return oss.str();
}

static bool validateAgeAtJoining(const std::string& dob, const std::string& joinDate) {
    int dobYear = parseYear(dob);
    int joinYear = parseYear(joinDate);
    return (joinYear - dobYear >= 20);
}

static bool validateDOBAge(const std::string& dob) {
    int dobYear = parseYear(dob);
    int currentYear = 2026;
    return (currentYear - dobYear >= 20);
}

static std::string readPassword() {
    std::string pass;
    while (true) {
        std::getline(std::cin, pass);
        if (pass.length() != 8) {
            std::cout << "Invalid. Password must be exactly 8 digits: ";
            continue;
        }
        bool onlyDigits = true;
        for (char c : pass) {
            if (c < '0' || c > '9') {
                onlyDigits = false;
                break;
            }
        }
        if (!onlyDigits) {
            std::cout << "Invalid. Password must contain numeric digits only: ";
            continue;
        }
        break;
    }
    return pass;
}

static int readInt() {
    int val;
    while (!(std::cin >> val)) {
        std::cin.clear();
        std::string dummy;
        std::getline(std::cin, dummy);
        std::cout << "Invalid. Please enter an integer: ";
    }
    flushInput();
    return val;
}

static double readDouble() {
    double val;
    while (!(std::cin >> val)) {
        std::cin.clear();
        std::string dummy;
        std::getline(std::cin, dummy);
        std::cout << "Invalid. Please enter a floating value: ";
    }
    flushInput();
    return val;
}

static std::string simpleHash(const std::string& str) {
    unsigned long hash = 5381;
    for (char c : str) {
        hash = ((hash << 5) + hash) + c;
    }
    return std::to_string(hash);
}

static bool isValidEmail(const std::string& email) {
    size_t atPos = email.find('@');
    size_t dotPos = email.rfind('.');
    if (atPos == std::string::npos || dotPos == std::string::npos) return false;
    if (atPos == 0 || dotPos == email.length() - 1 || dotPos < atPos + 2) return false;
    return true;
}

static bool isValidPhone(const std::string& phone) {
    if (phone.empty()) return false;
    size_t start = 0;
    if (phone[0] == '+') start = 1;
    if (phone.length() - start < 7 || phone.length() - start > 15) return false;
    for (size_t i = start; i < phone.length(); ++i) {
        if (phone[i] < '0' || phone[i] > '9') return false;
    }
    return true;
}

static std::string readContactInfo() {
    std::string contact;
    while (true) {
        std::cout << "Enter Contact Info (Email or Phone Number): ";
        contact = readString();
        if (isValidEmail(contact) || isValidPhone(contact)) {
            break;
        }
        std::cout << "Invalid contact format. Please input a valid Email (name@domain.com) or Phone (7-15 digits): " << std::endl;
    }
    return contact;
}

static bool isValidID(const std::string& id) {
    if (id.empty()) return false;
    for (char c : id) {
        if (!((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '-')) {
            return false;
        }
    }
    return true;
}

static std::string readValidID() {
    while (true) {
        std::string id = readString();
        if (isValidID(id)) {
            return id;
        }
        std::cout << "Invalid ID. Use alphanumeric characters and hyphens only: ";
    }
}

std::vector<std::string> splitString(const std::string& str, char delimiter) {
    std::vector<std::string> tokens;
    std::string token;
    std::istringstream tokenStream(str);
    while (std::getline(tokenStream, token, delimiter)) {
        if (!token.empty() && token.back() == '\r') {
            token.pop_back();
        }
        tokens.push_back(token);
    }
    return tokens;
}

// Change 1: loadEmployeesCSV — WarehouseManager no longer needs authorityLevel.
// extraField (tokens[8]) is kept as a placeholder 0 for WarehouseManager to preserve
// CSV column alignment; for StockKeeper it remains the assigned storage number.
void loadEmployeesCSV(SystemState& state) {
    std::ifstream file("employees.csv");
    if (!file.is_open()) return;
    std::string line;
    while (std::getline(file, line)) {
        if (line.empty() || line.find("ID,name") == 0) continue;
        auto tokens = splitString(line, ',');
        if (tokens.size() >= 9) {
            std::string id = tokens[0];
            std::string name = tokens[1];
            std::string role = tokens[2];
            std::string dob = tokens[3];
            std::string joinDate = tokens[4];
            std::string shift = tokens[5];
            double salary = std::stod(tokens[6]);
            double rate = std::stod(tokens[7]);
            int extra = std::stoi(tokens[8]); // 0 for WM, storageNum for SK
            int hours = 0;
            if (tokens.size() >= 10) {
                hours = std::stoi(tokens[9]);
            }
            if (role == "Warehouse Manager") {
                auto wm = std::make_unique<WarehouseManager>(id, name, salary, shift, dob, joinDate, rate);
                wm->setHoursWorked(hours);
                state.employees.push_back(std::move(wm));
            }
            else if (role == "Stock Keeper") {
                auto sk = std::make_unique<StockKeeper>(id, name, salary, shift, dob, joinDate, rate, extra);
                sk->setHoursWorked(hours);
                state.employees.push_back(std::move(sk));
            }
            else if (role == "Finance Manager") {
                auto fm = std::make_unique<FinanceManager>(id, name, salary, shift, dob, joinDate, rate);
                fm->setHoursWorked(hours);
                state.employees.push_back(std::move(fm));
            }
        }
    }
    file.close();
}

// Change 1: saveEmployeesCSV — WarehouseManager writes 0 as extraField placeholder.
void saveEmployeesCSV(const SystemState& state) {
    std::ofstream file("employees.csv", std::ios::trunc);
    if (!file.is_open()) return;
    file << "ID,name,role,DOB,joining date,shift,salary,rate,extraField,hoursWorked\n";
    for (const auto& emp : state.employees) {
        if (!emp) continue;
        int extra = 0;
        if (emp->getRole() == "Stock Keeper") {
            StockKeeper* sk = dynamic_cast<StockKeeper*>(emp.get());
            if (sk) extra = sk->getAssignedStorageNumber();
        }
        // WarehouseManager: extra stays 0
        file << emp->getEmployeeID() << "," << emp->getName() << "," << emp->getRole() << ","
            << emp->getDOB() << "," << emp->getJoiningDate() << "," << emp->getShiftSchedule() << ","
            << emp->getSalary() << "," << emp->getHourlyRate() << "," << extra << "," << emp->getHoursWorked() << "\n";
    }
    file.close();
}

void loadProductsCSV(SystemState& state) {
    std::ifstream file("products.csv");
    if (!file.is_open()) return;
    std::string line;
    while (std::getline(file, line)) {
        if (line.empty() || line.find("ID,name") == 0) continue;
        auto tokens = splitString(line, ',');
        if (tokens.size() >= 9) {
            std::string id = tokens[0];
            std::string name = tokens[1];
            double price = std::stod(tokens[2]);
            int qty = std::stoi(tokens[3]);
            std::string cat = tokens[4];
            std::string sub = tokens[5];
            int storageNum = std::stoi(tokens[6]);

            while (state.globalWarehouse.getStorageCount() < storageNum) {
                int nextID = state.globalWarehouse.getStorageCount() + 1;
                state.globalWarehouse.addStorage(Storage(nextID, 50, "Any"));
            }

            Barcode b(tokens[7], tokens[8]);
            try {
                if (cat == "Electronic") {
                    if (sub == "Standard" && tokens.size() >= 12) {
                        auto p = std::make_unique<ElectronicProduct>(id, name, price, qty, b, std::stoi(tokens[9]), std::stoi(tokens[10]), tokens[11]);
                        state.globalWarehouse.getStorage(storageNum - 1).addStock(std::move(p));
                    }
                    else if (sub == "Fragile" && tokens.size() >= 14) {
                        auto p = std::make_unique<FragileElectronics>(id, name, price, qty, b, std::stoi(tokens[9]), std::stoi(tokens[10]), tokens[11], std::stoi(tokens[12]), tokens[13]);
                        state.globalWarehouse.getStorage(storageNum - 1).addStock(std::move(p));
                    }
                }
                else if (cat == "Grocery") {
                    if (sub == "Standard" && tokens.size() >= 12) {
                        auto p = std::make_unique<GroceryProduct>(id, name, price, qty, b, std::stoi(tokens[9]), tokens[10] == "1", tokens[11]);
                        state.globalWarehouse.getStorage(storageNum - 1).addStock(std::move(p));
                    }
                    else if (sub == "VegFruit" && tokens.size() >= 12) {
                        auto p = std::make_unique<VegFruitGrocery>(id, name, price, qty, b, std::stoi(tokens[9]), tokens[10] == "1", tokens[11]);
                        state.globalWarehouse.getStorage(storageNum - 1).addStock(std::move(p));
                    }
                    else if (sub == "Perishable" && tokens.size() >= 14) {
                        auto p = std::make_unique<PerishableGrocery>(id, name, price, qty, b, std::stoi(tokens[9]), tokens[10] == "1", tokens[11], tokens[12], std::stod(tokens[13]));
                        state.globalWarehouse.getStorage(storageNum - 1).addStock(std::move(p));
                    }
                    else if (sub == "Non-Perishable" && tokens.size() >= 14) {
                        auto p = std::make_unique<NonPerishable>(id, name, price, qty, b, std::stoi(tokens[9]), tokens[10] == "1", tokens[11], std::stoi(tokens[12]), tokens[13]);
                        state.globalWarehouse.getStorage(storageNum - 1).addStock(std::move(p));
                    }
                }
                else if (cat == "Clothing") {
                    if (tokens.size() >= 12) {
                        auto p = std::make_unique<ClothingProduct>(id, name, price, qty, b, tokens[9], tokens[10], tokens[11]);
                        state.globalWarehouse.getStorage(storageNum - 1).addStock(std::move(p));
                    }
                }
            }
            catch (...) {}
        }
    }
    file.close();
}

void loadOrdersCSV(SystemState& state) {
    std::ifstream file("orders.csv");
    if (!file.is_open()) return;
    std::string line;
    while (std::getline(file, line)) {
        if (line.empty() || line.find("orderID") == 0) continue;
        auto tokens = splitString(line, ',');
        if (tokens.size() >= 7) {
            CustomerOrder ord;
            ord.orderID = tokens[0];
            ord.customerID = tokens[1];
            ord.productID = tokens[2];
            ord.quantity = std::stoi(tokens[3]);
            ord.totalPrice = std::stod(tokens[4]);
            ord.isApproved = (tokens[5] == "1");
            ord.isRejected = (tokens[6] == "1");
            state.pendingOrders.push_back(ord);
            if (ord.isApproved) {
                for (auto& cust : state.customers) {
                    if (cust.getCustomerID() == ord.customerID) {
                        Product* p = state.globalWarehouse.findProductByID(ord.productID);
                        PurchasedItem item;
                        item.productID = ord.productID;
                        item.productName = p ? p->getProductName() : "Unknown Asset";
                        item.quantity = ord.quantity;
                        item.pricePaid = ord.totalPrice;
                        item.purchaseDate = getCurrentDateTime().substr(0, 10);
                        item.warrantyMonths = p ? p->getWarrantyMonths() : 0;
                        cust.addPurchasedItem(item);
                        break;
                    }
                }
            }
        }
    }
    file.close();
}

void saveOrdersCSV(const SystemState& state) {
    std::ofstream file("orders.csv", std::ios::trunc);

    file << "orderID,customerID,productID,qty,totalPrice,isApproved,isRejected\n";
    for (const auto& ord : state.pendingOrders) {
        file << ord.orderID << "," << ord.customerID << "," << ord.productID << ","
            << ord.quantity << "," << ord.totalPrice << "," << (ord.isApproved ? 1 : 0) << ","
            << (ord.isRejected ? 1 : 0) << "\n";
    }
    file.close();
}

// ==========================================
// ZIP / XLSX BUILDER (CRC32 + ZIP format)
// ==========================================
static uint32_t _crc_table[256];
static bool _crc_ready = false;

static void _init_crc() {
    if (_crc_ready) return;
    for (uint32_t i = 0; i < 256; i++) {
        uint32_t c = i;
        for (int j = 0; j < 8; j++)
            c = (c & 1) ? (0xEDB88320u ^ (c >> 1)) : (c >> 1);
        _crc_table[i] = c;
    }
    _crc_ready = true;
}

static uint32_t _crc32s(const std::string& d) {
    _init_crc();
    uint32_t c = 0xFFFFFFFFu;
    for (unsigned char b : d)
        c = _crc_table[(c ^ b) & 0xFF] ^ (c >> 8);
    return c ^ 0xFFFFFFFFu;
}

struct _ZEntry {
    std::string name, data;
    uint32_t crc = 0, off = 0;
};

static void _pu16(std::vector<uint8_t>& b, uint16_t v) {
    b.push_back(v & 0xFF);
    b.push_back(v >> 8);
}

static void _pu32(std::vector<uint8_t>& b, uint32_t v) {
    b.push_back(v & 0xFF);
    b.push_back((v >> 8) & 0xFF);
    b.push_back((v >> 16) & 0xFF);
    b.push_back(v >> 24);
}

static void _pstr(std::vector<uint8_t>& b, const std::string& s) {
    for (char c : s) b.push_back((uint8_t)c);
}

static std::vector<uint8_t> _makeZip(std::vector<_ZEntry>& es) {
    std::vector<uint8_t> buf;
    for (auto& e : es) {
        e.off = (uint32_t)buf.size();
        e.crc = _crc32s(e.data);
        _pu32(buf, 0x04034b50); _pu16(buf, 20); _pu16(buf, 0); _pu16(buf, 0);
        _pu16(buf, 0); _pu16(buf, 0);
        _pu32(buf, e.crc);
        _pu32(buf, (uint32_t)e.data.size());
        _pu32(buf, (uint32_t)e.data.size());
        _pu16(buf, (uint16_t)e.name.size()); _pu16(buf, 0);
        _pstr(buf, e.name); _pstr(buf, e.data);
    }
    uint32_t cdOff = (uint32_t)buf.size();
    for (auto& e : es) {
        _pu32(buf, 0x02014b50); _pu16(buf, 20); _pu16(buf, 20);
        _pu16(buf, 0); _pu16(buf, 0); _pu16(buf, 0); _pu16(buf, 0);
        _pu32(buf, e.crc);
        _pu32(buf, (uint32_t)e.data.size());
        _pu32(buf, (uint32_t)e.data.size());
        _pu16(buf, (uint16_t)e.name.size()); _pu16(buf, 0); _pu16(buf, 0);
        _pu16(buf, 0); _pu16(buf, 0); _pu32(buf, 0); _pu32(buf, e.off);
        _pstr(buf, e.name);
    }
    uint32_t cdSz = (uint32_t)buf.size() - cdOff;
    _pu32(buf, 0x06054b50); _pu16(buf, 0); _pu16(buf, 0);
    _pu16(buf, (uint16_t)es.size()); _pu16(buf, (uint16_t)es.size());
    _pu32(buf, cdSz); _pu32(buf, cdOff); _pu16(buf, 0);
    return buf;
}

struct _SS {
    std::vector<std::string> strs;

    int add(const std::string& s) {
        for (int i = 0; i < (int)strs.size(); i++)
            if (strs[i] == s) return i;
        strs.push_back(s);
        return (int)strs.size() - 1;
    }

    static std::string escapeXml(const std::string& str) {
        std::ostringstream oss;
        for (char c : str) {
            switch (c) {
            case '&':  oss << "&amp;"; break;
            case '<':  oss << "&lt;"; break;
            case '>':  oss << "&gt;"; break;
            case '"':  oss << "&quot;"; break;
            case '\'': oss << "&apos;"; break;
            default:   oss << c; break;
            }
        }
        return oss.str();
    }

    std::string xml() const {
        std::ostringstream o;
        o << "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>"
            << "<sst xmlns=\"http://schemas.openxmlformats.org/spreadsheetml/2006/main\""
            << " count=\"" << strs.size() << "\" uniqueCount=\"" << strs.size() << "\">";
        for (auto& s : strs)
            o << "<si><t xml:space=\"preserve\">" << escapeXml(s) << "</t></si>";
        o << "</sst>";
        return o.str();
    }
};

static std::string _col(int c) {
    std::string r;
    while (c > 0) { r = (char)('A' + (c - 1) % 26) + r; c = (c - 1) / 26; }
    return r;
}

static std::string _ref(int r, int c) {
    return _col(c) + std::to_string(r);
}

static std::string _sc(int r, int c, int si, int s = 0) {
    std::ostringstream o;
    o << "<c r=\"" << _ref(r, c) << "\" t=\"s\" s=\"" << s << "\"><v>" << si << "</v></c>";
    return o.str();
}

static std::string _nc(int r, int c, double v, int s = 0) {
    std::ostringstream o;
    o << "<c r=\"" << _ref(r, c) << "\" s=\"" << s << "\"><v>"
        << std::fixed << std::setprecision(2) << v << "</v></c>";
    return o.str();
}

static std::string _fc(int r, int c, const std::string& f, int s = 0) {
    std::ostringstream o;
    o << "<c r=\"" << _ref(r, c) << "\" s=\"" << s << "\"><f>" << f << "</f></c>";
    return o.str();
}

        "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>"
        "<workbook xmlns=\"http://schemas.openxmlformats.org/spreadsheetml/2006/main\""
        " xmlns:r=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships\">"
        "<sheets>"
        "<sheet name=\"Treasury Summary\" sheetId=\"1\" r:id=\"rId1\"/>"
        "<sheet name=\"Sales Orders\"     sheetId=\"2\" r:id=\"rId2\"/>"
        "<sheet name=\"Payroll\"          sheetId=\"3\" r:id=\"rId3\"/>"
        "</sheets>"
        "</workbook>"
        });
    entries.push_back({ "xl/_rels/workbook.xml.rels",
        "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>"
        "<Relationships xmlns=\"http://schemas.openxmlformats.org/package/2006/relationships\">"
        "<Relationship Id=\"rId1\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet\"     Target=\"worksheets/sheet1.xml\"/>"
        "<Relationship Id=\"rId2\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet\"     Target=\"worksheets/sheet2.xml\"/>"
        "<Relationship Id=\"rId3\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet\"     Target=\"worksheets/sheet3.xml\"/>"

    entries.push_back({ "xl/styles.xml",            _stylesXml() });

    auto zip = _makeZip(entries);
    std::ofstream f("finance_management.xlsx", std::ios::binary);
    if (!f.is_open()) {
        std::cout << "[Error] Could not open finance_management.xlsx for writing." << std::endl;
        return;
    }
    f.write((char*)zip.data(), zip.size());
    f.close();

    std::cout << "[Excel Exported] finance_management.xlsx created successfully." << std::endl;
    std::cout << "  Sheet 1: Treasury Summary  (Capital, Net Worth, Liquidity check)" << std::endl;
    std::cout << "  Sheet 2: Sales Orders      (Per-order revenue, SUM total)" << std::endl;
    std::cout << "  Sheet 3: Payroll           (Wage = Rate x Hours x Shift factor)" << std::endl;
}

void runAdminPortal(SystemState& state) {
    if (performAdminLogin()) {
        while (true) {
            std::cout << "\n=== ADMINISTRATOR CONTROL PANEL ===" << std::endl;
            std::cout << "1. Register Employee Account" << std::endl;
            std::cout << "2. Register Supplier Account" << std::endl;
            std::cout << "3. View All System Data (Warehouse & Directory)" << std::endl;
            std::cout << "4. Audit Depot & Asset Shortages" << std::endl;
            std::cout << "5. Authorize/Send Employee Payroll List" << std::endl;
            std::cout << "6. Generate System Audit Log PDF Report" << std::endl;
            std::cout << "7. Log Out Admin" << std::endl;
            std::cout << "8. Delete Storage Section (and Reassign Products)" << std::endl;
            std::cout << "9. Create New Storage Section" << std::endl;
            std::cout << "Choice: ";
            int admChoice = readInt();
            if (admChoice == 7) break;

                // Change 1: No authority level prompt for any role.
                std::cout << "Select Role (1. Warehouse Manager, 2. Stock Keeper, 3. Finance Manager): ";
                int roleSel = readInt();

                if (roleSel == 3) {
                    int currentFin = 0;
                    for (const auto& emp : state.employees) {
                        if (emp && emp->getRole() == "Finance Manager") {
                            currentFin++;
                        }
                    }
                    if (currentFin >= 1) {
                        std::cout << "Error: Only one active Finance Manager account is allowed in the system." << std::endl;
                        continue;
                    }
                }

                std::cout << "Enter Employee ID (AlphaNumeric): ";
                std::string eID = readValidID();
                while (isEmployeeIDDuplicate(state, eID)) {
                    std::cout << "Error: Employee ID already registered. Use a unique ID: ";
                    eID = readValidID();
                }

                std::cout << "Enter Name (Alphabets Only): ";
                std::string eName = readAlphaString();

                std::cout << "Enter Date of Birth (YYYY-MM-DD): ";
                std::string dob = readDate();
                while (!validateDOBAge(dob)) {
                    std::cout << "\nError: Employee must be at least 20 years old relative to current year (2026).\n";
                    std::cout << "Re-enter Date of Birth (YYYY-MM-DD): ";
                    dob = readDate();
                }

                std::cout << "Enter Joining Date (YYYY-MM-DD): ";
                std::string joinDate = readDate();
                while (!validateAgeAtJoining(dob, joinDate)) {
                    std::cout << "\nError: Employee must be at least 20 years old at the time of joining." << std::endl;
                    std::cout << "Re-enter Joining Date (YYYY-MM-DD): ";
                    joinDate = readDate();
                }

                std::cout << "Enter Shift (Day/Night): ";
                std::string eShift = readAlphaString();

                double eSal = 0.0;
                if (roleSel == 1) {
                    // Change 1: Warehouse Manager — no authority level. Just salary and rate.
                    std::cout << "Enter Monthly Base Salary ($): ";
                    eSal = readDouble();
                    std::cout << "Enter Employee Hourly Wage Rate ($): ";
                    double rate = readDouble();
                    state.employees.push_back(std::make_unique<WarehouseManager>(eID, eName, eSal, eShift, dob, joinDate, rate));
                    state.systemLogs.recordAction("Registered Manager: " + eName);
                    std::cout << "Manager account configured." << std::endl;
                    saveEmployeesCSV(state);
                }
                else if (roleSel == 2) {
                    std::cout << "Assign to Primary Storage Section (1-" << state.globalWarehouse.getStorageCount() << "): ";
                    int storageNum = readInt();
                    std::cout << "Enter Monthly Base Salary ($): ";
                    eSal = readDouble();
                    std::cout << "Enter Employee Hourly Wage Rate ($): ";
                    double rate = readDouble();
                    state.employees.push_back(std::make_unique<StockKeeper>(eID, eName, eSal, eShift, dob, joinDate, rate, storageNum));
                    state.systemLogs.recordAction("Registered StockKeeper: " + eName);
                    std::cout << "Stock Keeper account configured." << std::endl;
                    saveEmployeesCSV(state);
                }
                else if (roleSel == 3) {
                    std::cout << "Enter Monthly Base Salary ($): ";
                    eSal = readDouble();
                    std::cout << "Enter Employee Hourly Wage Rate ($15 - $150): ";
                    double rate = readDouble();
                    while (rate < 15.0 || rate > 150.0) {
                        std::cout << "Invalid range. Enter hourly rate between $15 and $150: ";
                        rate = readDouble();
                    }
                    state.employees.push_back(std::make_unique<FinanceManager>(eID, eName, eSal, eShift, dob, joinDate, rate));
                    state.systemLogs.recordAction("Registered Finance Manager: " + eName);
                    std::cout << "Finance Manager account configured." << std::endl;
                    saveEmployeesCSV(state);
                }
                else {
                    std::cout << "Invalid role choice." << std::endl;
                }
            }
            else if (admChoice == 2) {
                std::cout << "Enter Supplier ID (AlphaNumeric): ";
                std::string sID = readValidID();
                while (isSupplierIDDuplicate(state, sID)) {
                    std::cout << "Error: Supplier ID already registered. Use a unique ID: ";
                    sID = readValidID();
                }
                std::cout << "Enter Supplier Corporate Name (Alphabets Only): ";
                std::string sName = readAlphaString();
                std::cout << "Enter Contract Terms (e.g. Net 30): ";
                std::string sTerms = readString();
                state.suppliers.push_back(Supplier(sID, sName, sTerms));
                state.systemLogs.recordAction("Registered Supplier: " + sName);
                std::cout << "Supplier registered successfully!" << std::endl;
                saveSuppliersCSV(state);
            }
            else if (admChoice == 3) {
                std::cout << "\n--- GLOBAL WAREHOUSE CONFIGURATION ---" << std::endl;
                std::cout << state.globalWarehouse;
                std::cout << "\n--- REGISTERED STAFF DIRECTORY ---" << std::endl;
                if (state.employees.empty()) std::cout << "  No active employee accounts configured." << std::endl;
                for (auto& emp : state.employees) {
                    if (emp) emp->generateReport();
                }
                std::cout << "\n--- REGISTERED SUPPLIER DIRECTORY ---" << std::endl;
                for (const auto& sup : state.suppliers) {
                    std::cout << "  * " << sup << std::endl;
                }
                std::cout << "\n--- REGISTERED CUSTOMER DIRECTORY ---" << std::endl;
                if (state.customers.empty()) std::cout << "  No customer accounts registered." << std::endl;
                for (const auto& cust : state.customers) {
                    std::cout << "  * " << cust << std::endl;
                }
                std::cout << "\n--- SYSTEM AUDIT LOGS ---" << std::endl;
                state.systemLogs.printAuditTrail();
            }
            else if (admChoice == 4) {
                state.globalWarehouse.findShortages();
            }
            else if (admChoice == 5) {
                state.payrollAuthorizedByAdmin = true;
                state.systemLogs.recordAction("Admin sent employee list to Treasury (Authorized next payroll).");
                std::cout << "SUCCESS: Employee list authorized and sent to Treasury." << std::endl;
            }
            else if (admChoice == 6) {
                generateAuditLogReport(state);
            }
            else if (admChoice == 8) {
                std::cout << "\n--- DECOMMISSION STORAGE SECTION ---" << std::endl;
                std::cout << "Enter Storage Number to delete: ";
                int delID = readInt();

                int delIdx = -1;
                for (int i = 0; i < state.globalWarehouse.getStorageCount(); ++i) {
                    if (state.globalWarehouse.getStorage(i).getStorageNumber() == delID) {
                        delIdx = i; break;
                    }
                }

                if (delIdx == -1) {
                    std::cout << "Error: Storage section " << delID << " not found." << std::endl;
                }
                else {
                    Storage& sectionToDelete = state.globalWarehouse.getStorage(delIdx);
                    bool canDelete = true;

                    if (sectionToDelete.getProductCount() > 0) {
                        std::cout << "WARNING: Section contains " << sectionToDelete.getProductCount() << " items." << std::endl;
                        std::cout << "1. Move items to an existing section\n"
                            << "2. Create a new section and move items there\n"
                            << "3. Cancel deletion\nChoice: ";
                        int subChoice = readInt();

                        if (subChoice == 1 || subChoice == 2) {
                            int tIdx = -1;

                            if (subChoice == 2) {
                                std::cout << "--- CREATE NEW STORAGE SECTION (inline) ---" << std::endl;
                                std::cout << "New Storage Number: ";
                                int nID = readInt();
                                std::cout << "Capacity: ";
                                int nCap = readInt();
                                std::cout << "Category (Electronic, Clothing, Grocery, Any): ";
                                std::string nCat = readString();
                                state.globalWarehouse.addStorage(Storage(nID, nCap, nCat));
                                state.systemLogs.recordAction("Admin created inline Section: " + std::to_string(nID));
                                std::cout << "New Section " << nID << " created. Items will be moved there." << std::endl;
                                tIdx = state.globalWarehouse.getStorageCount() - 1;
                            }
                            else {
                                std::cout << "Enter Target Storage Number to move items to: ";
                                int tID = readInt();
                                for (int i = 0; i < state.globalWarehouse.getStorageCount(); ++i) {
                                    if (state.globalWarehouse.getStorage(i).getStorageNumber() == tID) {
                                        tIdx = i; break;
                                    }
                                }
                            }

                            if (tIdx != -1 && tIdx != delIdx) {
                                Storage& target = state.globalWarehouse.getStorage(tIdx);
                                Storage& src = state.globalWarehouse.getStorage(delIdx);
                                while (src.getProductCount() > 0) {
                                    if (target.getProductCount() < target.getCapacity()) {
                                        auto p = src.takeProduct(0);
                                        if (!target.addStock(std::move(p))) {
                                            std::cout << "Category Mismatch! Cannot move all items to target section." << std::endl;
                                            canDelete = false; break;
                                        }
                                    }
                                    else {
                                        std::cout << "Target storage is FULL. Move failed." << std::endl;
                                        canDelete = false; break;
                                    }
                                }
                                if (canDelete) {
                                    std::cout << "All items shifted to Storage "
                                        << state.globalWarehouse.getStorage(tIdx).getStorageNumber()
                                        << " successfully." << std::endl;
                                }
                            }
                            else {
                                std::cout << "Invalid target section." << std::endl;
                                canDelete = false;
                            }
                        }
                        else {
                            canDelete = false;
                        }
                    }

                    if (canDelete) {
                        int finalDelIdx = -1;
                        for (int i = 0; i < state.globalWarehouse.getStorageCount(); ++i) {
                            if (state.globalWarehouse.getStorage(i).getStorageNumber() == delID) {
                                finalDelIdx = i; break;
                            }
                        }
                        if (finalDelIdx != -1) {
                            state.globalWarehouse.removeStorageByIndex(finalDelIdx);
                            state.systemLogs.recordAction("Admin deleted Storage " + std::to_string(delID));
                            saveAllCSVFiles(state);
                            std::cout << "Storage Section " << delID << " has been removed." << std::endl;
                        }
                    }
                }
            }
            else if (admChoice == 9) {
                std::cout << "\n--- CREATE NEW STORAGE SECTION ---" << std::endl;
                std::cout << "New Storage Number: "; int nID = readInt();
                std::cout << "Capacity: "; int nCap = readInt();
                std::cout << "Category (Electronic, Clothing, Grocery, Any): ";
                std::string nCat = readString();
                state.globalWarehouse.addStorage(Storage(nID, nCap, nCat));
                state.systemLogs.recordAction("Admin created new Section: " + std::to_string(nID));
                saveAllCSVFiles(state);
                std::cout << "SUCCESS: New section is now active." << std::endl;
            }
        }
    }
}

    for (auto& emp : state.employees) {
        if (emp && emp->getEmployeeID() == eID) {
            activeEmp = emp.get();
            break;
        }
    }
    if (!activeEmp) {
        std::cout << "No matching ID found. Admin registration required.\n" << std::endl;
        return;
    }

    WarehouseManager* manager = dynamic_cast<WarehouseManager*>(activeEmp);
    StockKeeper* stockKeeper = dynamic_cast<StockKeeper*>(activeEmp);
    FinanceManager* finance = dynamic_cast<FinanceManager*>(activeEmp);

    // =========================================================
    // Change 1: Single unified Warehouse Manager portal menu.
    // All permissions available; no sub-level tiering.
    // =========================================================
    if (manager) {
        while (true) {
            std::cout << "\n=== WAREHOUSE MANAGER PORTAL ===" << std::endl;
            std::cout << "1. Assign Routine Task" << std::endl;
            std::cout << "2. Audit Depot & Asset Shortages / Generate Reports" << std::endl;
            std::cout << "3. Request Product Restock (via Supplier)" << std::endl;
            std::cout << "4. View Global System Logs" << std::endl;
            std::cout << "5. View My Profile Info" << std::endl;
            std::cout << "6. Logout" << std::endl;
            std::cout << "Choice: ";
            int mgrChoice = readInt();
            if (mgrChoice == 6) break;

            if (mgrChoice == 1) {
                manager->assignTask();
            }
            else if (mgrChoice == 2) {
                manager->auditWarehouse(state.globalWarehouse);
                generateWarehouseReport(state);
            }
            else if (mgrChoice == 3) {
                // Restock request logic — no dollar cap (Change 1).
                std::cout << "Enter the ID of the product to restock: ";
                std::string pID = readString();
                Product* p = state.globalWarehouse.findProductByID(pID);
                if (!p) {
                    std::cout << "Product not found." << std::endl;
                    continue;
                }
                std::cout << "Enter quantity to request: ";
                int reqQty = readInt();
                if (state.suppliers.empty()) {
                    std::cout << "No suppliers registered. Ask Admin to add a supplier first." << std::endl;
                    continue;
                }
                std::cout << "Available Suppliers:\n";
                for (size_t i = 0; i < state.suppliers.size(); ++i) {
                    std::cout << "  [" << i + 1 << "] ID: " << state.suppliers[i].getSupplierID()
                        << " Name: " << state.suppliers[i].getSupplierName() << std::endl;
                }
                std::cout << "Select Supplier Index: ";
                int supIdx = readInt();
                if (supIdx < 1 || supIdx > static_cast<int>(state.suppliers.size())) {
                    std::cout << "Invalid index." << std::endl;
                    continue;
                }
                RestockRequest req;
                std::ostringstream reqID;
                reqID << "ORD-" << state.restockRequests.size() + 100;
                req.orderID = reqID.str();
                req.supplierID = state.suppliers[supIdx - 1].getSupplierID();
                req.productID = pID;
                req.quantityRequested = reqQty;
                req.isFulfilled = false;
                state.restockRequests.push_back(req);
                state.systemLogs.recordAction("Restock request " + req.orderID + " created by Manager " + manager->getName() + ".");
                std::cout << "SUCCESS: Restock request " << req.orderID << " submitted to supplier." << std::endl;
                saveRestockRequestsCSV(state);
            }
            else if (mgrChoice == 4) {
                state.systemLogs.printAuditTrail();
            }
            else if (mgrChoice == 5) {
                manager->generateReport();

                PerishableGrocery* pg = dynamic_cast<PerishableGrocery*>(p);
                if (pg) {
                    stockKeeper->flagExpiredItem(*pg);
                }
                else {
                    std::cout << "Product is not classifiable as Perishable Grocery." << std::endl;
                }
            }

                std::cout << "Enter Product ID: ";
                std::string targetID = readString();
                Product* p = state.globalWarehouse.findProductByID(targetID);
                if (p) {
                    std::cout << "Current quantity: " << p->getQuantity() << ". Enter restocking units: ";
                    int restockUnits = readInt();
                    stockKeeper->restockItem(p, restockUnits);
                    saveProductsCSV(state);
                }
                else {
                    std::cout << "Product not found." << std::endl;
                }
            }
            else if (skChoice == 5) {
                std::cout << "Enter Product ID to apply discount: ";
                std::string targetID = readString();
                Product* p = state.globalWarehouse.findProductByID(targetID);
                if (!p) {
                    std::cout << "Product not found." << std::endl;
                    continue;
                }
                std::cout << "Select Discount Type:\n  1. Bulk Discount\n  2. Seasonal Discount\nChoice: ";

                std::string polName = readAlphaString();

                std::string today = getCurrentDateTime().substr(0, 10);
                while (true) {
                    std::cout << "Enter Start Date (YYYY-MM-DD) [Today or future]: ";
                    startDate = readDate();
                    if (!isDateBeforeOrEqual(today, startDate)) {
                        std::cout << "Error: Start date cannot be in the past (Today is " << today << "). Please re-enter.\n";
                        continue;
                    }
                    break;
                }
                while (true) {
                    std::cout << "Enter End Date (YYYY-MM-DD) [Must be >= Start Date]: ";
                    endDate = readDate();
                    if (!isDateBeforeOrEqual(startDate, endDate)) {
                        std::cout << "Error: End date cannot be before start date (" << startDate << "). Please re-enter.\n";
                        continue;
                    }
                    break;
                }

                    std::cout << "Enter Minimum Quantity for Bulk Discount: ";
                    int minQty = readInt();
                    std::cout << "Enter Discount Percentage (e.g., 15.0 for 15%): ";
                    double discPct = readDouble();
                    BulkDiscount policy(polName, startDate, endDate, minQty, discPct);
                    p->applyDiscount(policy);
                    std::cout << "Bulk Discount applied. Updated Product Details:\n";
                    p->displayStatus();
                }

                    std::cout << "Enter Season Name (e.g., Summer): ";
                    std::string sName = readAlphaString();
                    std::cout << "Enter Discount Rate (e.g., 0.10 for 10%): ";
                    double rate = readDouble();
                    SeasonalDiscount policy(polName, startDate, endDate, sName, rate);
                    p->applyDiscount(policy);
                    std::cout << "Seasonal Discount applied. Updated Product Details:\n";
                    p->displayStatus();

            std::cout << "1. Run Treasury & Inventory Money health Audit" << std::endl;
            std::cout << "2. Record Operational Expense transaction" << std::endl;
            std::cout << "3. Calculate & Pay Monthly Employee Wages (Interactive)" << std::endl;
            std::cout << "4. Process Pending Customer Orders" << std::endl;
            std::cout << "5. Process Warranty Claim Tickets (Replace / Refund)" << std::endl;
            std::cout << "6. Export Smart Finance Excel Worksheet (CSV)" << std::endl;
            std::cout << "7. View My Profile Info" << std::endl;
            std::cout << "8. Logout" << std::endl;
            std::cout << "Choice: ";
            int finChoice = readInt();
            if (finChoice == 8) break;

                finance->auditTreasury(state.globalWarehouse, state.availableCapital, state.totalOperatingExpenses);

            else if (finChoice == 2) {
                std::cout << "Enter expense transaction amount ($): ";
                double amt = readDouble();
                finance->addOperationalExpense(amt, state.availableCapital, state.totalOperatingExpenses);
                state.systemLogs.recordAction("Logged operating expense transaction of $" + std::to_string(amt));
                saveAllCSVFiles(state);
            }

                if (!state.payrollAuthorizedByAdmin) {
                    std::cout << "\nError: Payroll has already been processed for this cycle.\n"
                        << "Wages cannot be calculated until the Administrator sends the updated employee list." << std::endl;
                    continue;
                }
                std::cout << "\n--- AUDITING CORPORATE PAYROLL (Interactive) ---" << std::endl;
                double totalPayroll = 0.0;
                if (state.employees.empty()) {
                    std::cout << "  No active employees registered to pay." << std::endl;
                }
                else {
                    for (auto& emp : state.employees) {
                        if (!emp) continue;
                        double wage = 0.0;
                        if (emp.get() == finance) {
                            wage = emp->calculateMonthlyWage();
                            totalPayroll += wage;
                            std::cout << "  * Employee: " << emp->getName()
                                << " (ID: " << emp->getEmployeeID() << " | " << emp->getRole() << ")\n"
                                << "    -> Paid standard monthly flat contract salary: $" << wage << std::endl;
                        }

                            std::cout << "  * Enter hours worked this month for " << emp->getName()
                                << " (ID: " << emp->getEmployeeID() << " | " << emp->getRole() << "): ";
                            int hrs = readInt();
                            while (hrs < 0 || hrs > 400) {
                                std::cout << "Invalid hours. Enter a value between 0 and 400: ";
                                hrs = readInt();
                            }
                            emp->setHoursWorked(hrs);
                            wage = emp->calculateMonthlyWage();
                            totalPayroll += wage;
                            std::cout << "    -> Calculated Wage: $" << wage << " (" << hrs
                                << " hrs @ $" << emp->getHourlyRate() << "/hr | Shift: " << emp->getShiftSchedule() << ")\n";
                        }
                    }
                    std::cout << "\nTotal Gross Payroll Expenses: $" << totalPayroll << std::endl;
                    if (state.availableCapital >= totalPayroll) {
                        state.availableCapital -= totalPayroll;
                        state.totalOperatingExpenses += totalPayroll;
                        std::cout << "SUCCESS: Paid monthly salaries. Capital updated successfully." << std::endl;
                        state.systemLogs.recordAction("Processed payroll transaction of $" + std::to_string(totalPayroll));
                        state.payrollAuthorizedByAdmin = false;
                        generatePayrollReport(state);
                        saveAllCSVFiles(state);
                    }
                    else {
                        std::cout << "ERROR: Insufficient available capital to process payroll!" << std::endl;
                    }
                }
            }
            else if (finChoice == 4) {
                std::cout << "\n--- PENDING CUSTOMER ORDERS ---" << std::endl;
                bool found = false;
                for (const auto& ord : state.pendingOrders) {
                    if (!ord.isApproved && !ord.isRejected) {
                        std::cout << "  * Order ID: " << ord.orderID
                            << " | Customer ID: " << ord.customerID
                            << " | Product ID: " << ord.productID
                            << " | Qty: " << ord.quantity
                            << " | Total Price: $" << ord.totalPrice << std::endl;
                        found = true;
                    }
                }
                if (!found) {
                    std::cout << "  No pending customer orders found." << std::endl;
                    continue;
                }
                std::cout << "Enter Order ID to process: ";
                std::string targetOrdID = readString();
                CustomerOrder* selectedOrd = nullptr;
                for (auto& ord : state.pendingOrders) {
                    if (ord.orderID == targetOrdID && !ord.isApproved && !ord.isRejected) {
                        selectedOrd = &ord;
                        break;
                    }
                }
                if (!selectedOrd) {
                    std::cout << "Invalid or already processed Order ID." << std::endl;
                    continue;
                }
                std::cout << "Select Action (1 to Approve, 2 to Reject): ";
                int action = readInt();
                if (action == 1) {
                    Product* p = state.globalWarehouse.findProductByID(selectedOrd->productID);
                    if (p && p->getQuantity() >= selectedOrd->quantity) {
                        p->setQuantity(p->getQuantity() - selectedOrd->quantity);
                        state.availableCapital += selectedOrd->totalPrice;
                        std::string currentCustName = "Guest";
                        for (auto& cust : state.customers) {
                            if (cust.getCustomerID() == selectedOrd->customerID) {
                                PurchasedItem item;
                                item.productID = selectedOrd->productID;
                                item.productName = p->getProductName();
                                item.quantity = selectedOrd->quantity;
                                item.pricePaid = selectedOrd->totalPrice;
                                item.purchaseDate = getCurrentDateTime().substr(0, 10);
                                item.warrantyMonths = p->getWarrantyMonths();
                                cust.addPurchasedItem(item);
                                currentCustName = cust.getName();
                                break;
                            }
                        }
                        selectedOrd->isApproved = true;
                        state.systemLogs.recordAction("Finance Manager approved customer order " + selectedOrd->orderID);
                        std::cout << "SUCCESS: Order approved. Stock deducted, corporate funds updated, items granted." << std::endl;
                        generateCustomerReceipt(*selectedOrd, currentCustName);
                        saveOrdersCSV(state);
                        saveProductsCSV(state);
                        saveCustomersCSV(state);
                        generateOrdersReport(state);
                    }
                    else {
                        std::cout << "Failed: Insufficient warehouse stock to fulfill this order. Auto-rejecting." << std::endl;
                        selectedOrd->isRejected = true;
                        saveOrdersCSV(state);
                    }
                }
                else if (action == 2) {
                    selectedOrd->isRejected = true;
                    state.systemLogs.recordAction("Finance Manager rejected customer order " + selectedOrd->orderID);
                    std::cout << "Order successfully rejected." << std::endl;
                    saveOrdersCSV(state);
                }
            }
            else if (finChoice == 5) {
                std::cout << "\n--- WARRANTY CLAIMS QUEUE ---" << std::endl;
                bool found = false;
                for (const auto& cl : state.warrantyClaims) {
                    if (!cl.isProcessed) {
                        std::cout << "  * Claim ID: " << cl.claimID
                            << " | Customer ID: " << cl.customerID
                            << " | Product ID: " << cl.productID
                            << " | Type: " << cl.claimType
                            << " | Refund Value: $" << cl.refundAmount << std::endl;
                        found = true;
                    }
                }
                if (!found) {
                    std::cout << "  No pending warranty claims found." << std::endl;
                    continue;
                }
                std::cout << "Enter Claim ID to process: ";
                std::string targetClaimID = readString();
                WarrantyClaim* selectedCl = nullptr;
                for (auto& cl : state.warrantyClaims) {
                    if (cl.claimID == targetClaimID && !cl.isProcessed) {
                        selectedCl = &cl;
                        break;
                    }
                }
                if (!selectedCl) {
                    std::cout << "Invalid or already processed Claim ID." << std::endl;
                    continue;
                }
                if (selectedCl->claimType == "Replacement") {
                    Product* p = state.globalWarehouse.findProductByID(selectedCl->productID);
                    if (p && p->getQuantity() >= 1) {
                        p->setQuantity(p->getQuantity() - 1);
                        selectedCl->isProcessed = true;
                        state.systemLogs.recordAction("Approved warranty claim replacement " + selectedCl->claimID);
                        std::cout << "SUCCESS: Replacement item approved and deducted from warehouse stock." << std::endl;
                        saveWarrantyClaimsCSV(state);
                        saveProductsCSV(state);
                        generateOrdersReport(state);
                    }
                    else {
                        std::cout << "Failed: Product is out of stock. Suggest a refund instead." << std::endl;
                    }
                }
                else if (selectedCl->claimType == "Refund") {
                    if (state.availableCapital >= selectedCl->refundAmount) {
                        state.availableCapital -= selectedCl->refundAmount;
                        state.totalOperatingExpenses += selectedCl->refundAmount;
                        selectedCl->isProcessed = true;
                        state.systemLogs.recordAction("Approved warranty claim refund " + selectedCl->claimID);
                        std::cout << "SUCCESS: Refund approved. $" << selectedCl->refundAmount << " paid from capital." << std::endl;
                        saveWarrantyClaimsCSV(state);
                        generateOrdersReport(state);
                    }
                    else {
                        std::cout << "Failed: Insufficient corporate capital to issue refund." << std::endl;
                    }
                }
            }
            else if (finChoice == 6) {
                generateFinanceExcel(state);
            }
            else if (finChoice == 7) {
                finance->generateReport();
            }
        }
    }
}

void runCustomerPortal(SystemState& state) {
    std::cout << "=== CLIENT PORTAL ===" << std::endl;
    std::cout << "1. Sign In (Registered Customer)" << std::endl;
    std::cout << "2. Sign Up (New Registration)" << std::endl;
    std::cout << "Choice: ";
    int custPortalSel = readInt();

    Customer* activeCust = nullptr;
    if (custPortalSel == 1) {
        std::cout << "Enter Customer ID: ";
        std::string cID = readString();
        for (auto& cust : state.customers) {
            if (cust.getCustomerID() == cID) {
                activeCust = &cust;
                break;
            }
        }
        if (!activeCust) {
            std::cout << "ID not verified in staff system directories.\n" << std::endl;
            return;
        }
    }
    else if (custPortalSel == 2) {
        std::cout << "Enter Unique Customer ID: ";
        std::string cID = readValidID();
        while (isCustomerIDDuplicate(state, cID)) {
            std::cout << "Error: Customer ID already registered. Use a unique ID: ";
            cID = readValidID();
        }
        std::cout << "Enter Your Name (Alphabets Only): ";
        std::string cName = readAlphaString();
        std::string cContact = readContactInfo();
        state.customers.push_back(Customer(cID, cName, cContact));
        activeCust = &state.customers.back();
        std::cout << "Registration successful! Welcome " << cName << "!\n";
        saveCustomersCSV(state);
    }
    else {
        return;
    }

    while (true) {
        std::cout << "\n=== WELCOME TO THE SHOPPING MENU ===" << std::endl;
        std::cout << "1. Browse Store Products" << std::endl;
        std::cout << "2. Purchase Item (Submits pending order)" << std::endl;
        std::cout << "3. View My Purchases Trail" << std::endl;
        std::cout << "4. File Warranty Claim (Replace or Refund)" << std::endl;
        std::cout << "5. Logout" << std::endl;
        std::cout << "Choice: ";
        int shopChoice = readInt();
        if (shopChoice == 5) break;

        if (shopChoice == 1) {
            std::cout << "\n--- BROWSE CATALOG ---" << std::endl;
            std::cout << state.globalWarehouse;
        }
        else if (shopChoice == 2) {
            std::cout << "Enter the Product ID you wish to purchase: ";
            std::string targetID = readString();
            Product* item = state.globalWarehouse.findProductByID(targetID);

            if (!item) {
                std::cout << "Error: Product ID does not exist in store." << std::endl;
            }
            else {
                if (item->getSubtype() == "VegFruit") {
                    std::cout << "This Veg/Fruit item is sold by weight ($" << item->getPrice() << "/kg).\n";
                    std::cout << "Select purchase unit:\n  1. Kilograms (kg)\n  2. Grams (g)\nChoice: ";
                    int unitSel = readInt();
                    double targetGrams = 0.0;
                    double calculatedCost = 0.0;
                    if (unitSel == 1) {
                        std::cout << "Enter weight in Kilograms (e.g. 1.5): ";
                        double kgVal = readDouble();
                        targetGrams = kgVal * 1000.0;
                        calculatedCost = item->getPrice() * kgVal;
                    }
                    else {
                        std::cout << "Enter weight in Grams (e.g. 450): ";
                        double gVal = readDouble();
                        targetGrams = gVal;
                        calculatedCost = (item->getPrice() / 1000.0) * gVal;
                    }
                    int orderQtyGrams = static_cast<int>(targetGrams);
                    if (item->getQuantity() >= orderQtyGrams) {
                        CustomerOrder ord;
                        std::ostringstream ordID;
                        ordID << "CUST-ORD-" << state.pendingOrders.size() + 100;
                        ord.orderID = ordID.str();
                        ord.customerID = activeCust->getCustomerID();
                        ord.productID = targetID;
                        ord.quantity = orderQtyGrams;
                        ord.totalPrice = calculatedCost;
                        ord.isApproved = false;
                        ord.isRejected = false;
                        state.pendingOrders.push_back(ord);
                        state.systemLogs.recordAction("Customer " + activeCust->getCustomerName() + " submitted pending Veg/Fruit order " + ord.orderID);
                        std::cout << "[Order Received] Order " << ord.orderID << " submitted successfully.\n"
                            << "  -> Weight requested: " << (targetGrams / 1000.0) << " kg | Total: $" << calculatedCost << std::endl;
                        saveOrdersCSV(state);
                    }
                    else {
                        double currentKgAvailable = static_cast<double>(item->getQuantity()) / 1000.0;
                        std::cout << "[Failed] Insufficient stock. Only " << currentKgAvailable << " kg available." << std::endl;
                    }
                }
                else {
                    std::cout << "Product: " << item->getProductName() << " | Price: $" << item->getPrice() << std::endl;
                    std::cout << "Enter quantity: ";
                    int orderQty = readInt();
                    if (item->getQuantity() >= orderQty) {
                        CustomerOrder ord;
                        std::ostringstream ordID;
                        ordID << "CUST-ORD-" << state.pendingOrders.size() + 100;
                        ord.orderID = ordID.str();
                        ord.customerID = activeCust->getCustomerID();
                        ord.productID = targetID;
                        ord.quantity = orderQty;
                        ord.totalPrice = item->getPrice() * orderQty;
                        ord.isApproved = false;
                        ord.isRejected = false;
                        state.pendingOrders.push_back(ord);
                        state.systemLogs.recordAction("Customer " + activeCust->getCustomerName() + " submitted pending order " + ord.orderID);
                        std::cout << "[Order Received] Order " << ord.orderID << " submitted successfully.\n"
                            << "  -> Awaiting Finance Manager approval. Stock is not yet deducted." << std::endl;
                        saveOrdersCSV(state);
                    }
                    else {
                        std::cout << "[Failed] Insufficient stock. Only " << item->getQuantity() << " available." << std::endl;
                    }
                }
            }
        }
        else if (shopChoice == 3) {
            activeCust->viewPurchaseHistory();
            std::cout << "Total Spent: $" << activeCust->getTotalSpent() << std::endl;
        }
        else if (shopChoice == 4) {
            std::cout << "\n--- FILE WARRANTY CLAIM ---" << std::endl;
            const auto& purchases = activeCust->getPurchasedItems();
            if (purchases.empty()) {
                std::cout << "You have no approved purchases eligible for warranty claims." << std::endl;
                continue;
            }
            std::cout << "Eligible Purchases:\n";
            for (size_t i = 0; i < purchases.size(); ++i) {
                std::cout << "  [" << i + 1 << "] " << purchases[i].productName
                    << " (ID: " << purchases[i].productID
                    << ") | Purchased: " << purchases[i].purchaseDate
                    << " | Warranty: " << purchases[i].warrantyMonths << " Months" << std::endl;
            }
            std::cout << "Select item to claim (Index): ";
            int idx = readInt();
            if (idx < 1 || idx > static_cast<int>(purchases.size())) {
                std::cout << "Invalid index selection." << std::endl;
                continue;
            }
            const auto& selected = purchases[idx - 1];
            if (selected.warrantyMonths == 0) {
                std::cout << "This product does not have an active warranty." << std::endl;
                continue;
            }
            std::string today = getCurrentDateTime().substr(0, 10);
            std::string warrantyEndDate = addMonthsToDate(selected.purchaseDate, selected.warrantyMonths);
            if (today > warrantyEndDate) {
                std::cout << "Warranty expired. (Valid until: " << warrantyEndDate << " | Today: " << today << ")" << std::endl;
                continue;
            }
            std::cout << "Warranty is VALID until: " << warrantyEndDate << std::endl;
            std::cout << "Select Option:\n  1. Replacement product\n  2. Refund (Money Back)\nChoice: ";
            int claimSel = readInt();
            if (claimSel != 1 && claimSel != 2) {
                std::cout << "Invalid selection." << std::endl;
                continue;
            }
            WarrantyClaim claim;
            std::ostringstream claimID;
            claimID << "CLAIM-" << state.warrantyClaims.size() + 100;
            claim.claimID = claimID.str();
            claim.customerID = activeCust->getCustomerID();
            claim.productID = selected.productID;
            claim.claimType = (claimSel == 1) ? "Replacement" : "Refund";
            claim.refundAmount = selected.pricePaid;
            claim.isProcessed = false;
            state.warrantyClaims.push_back(claim);
            state.systemLogs.recordAction("Customer " + activeCust->getCustomerName() + " filed warranty claim " + claim.claimID);
            std::cout << "SUCCESS: Warranty claim " << claim.claimID << " submitted. Awaiting Finance Manager review." << std::endl;
            saveWarrantyClaimsCSV(state);
        }
    }
}

            break;
        }

        return;
    }

    while (true) {
        std::cout << "\n=== SUPPLIER DASHBOARD: " << activeSup->getSupplierName() << " ===" << std::endl;
        std::cout << "1. View Pending Deliveries" << std::endl;
        std::cout << "2. Fulfill Pending Delivery Order" << std::endl;
        std::cout << "3. View Contract Terms & Invoice Summary" << std::endl;
        std::cout << "4. Logout" << std::endl;
        std::cout << "Choice: ";
        int supChoice = readInt();
        if (supChoice == 4) break;

        if (supChoice == 1) {
            std::cout << "\n--- PENDING DELIVERY SLOTS ---" << std::endl;
            bool found = false;
            for (const auto& req : state.restockRequests) {
                if (req.supplierID == activeSup->getSupplierID() && !req.isFulfilled) {
                    std::cout << "  * Order ID: " << req.orderID
                        << " | Product ID: " << req.productID
                        << " | Quantity Requested: " << req.quantityRequested << std::endl;
                    found = true;
                }
            }
            if (!found) {
                std::cout << "  No pending requests assigned from depot managers." << std::endl;
            }
        }
        else if (supChoice == 2) {
            std::cout << "Enter the Pending Order ID you wish to fulfill: ";
            std::string ordID = readString();
            int targetIdx = -1;
            for (size_t i = 0; i < state.restockRequests.size(); ++i) {
                if (state.restockRequests[i].orderID == ordID &&
                    state.restockRequests[i].supplierID == activeSup->getSupplierID() &&
                    !state.restockRequests[i].isFulfilled) {
                    targetIdx = static_cast<int>(i);
                    break;
                }
            }
            if (targetIdx == -1) {
                std::cout << "Active, unfulfilled Order ID matching criteria not located." << std::endl;
            }
            else {
                Product* p = state.globalWarehouse.findProductByID(state.restockRequests[targetIdx].productID);
                if (p) {
                    activeSup->orderRestock(*p, state.restockRequests[targetIdx].quantityRequested);
                    state.restockRequests[targetIdx].isFulfilled = true;
                    state.systemLogs.recordAction("Fulfilled Restock Order: " + ordID);
                    std::cout << "Delivery completed. Products updated inside warehouse stock." << std::endl;
                    saveRestockRequestsCSV(state);
                    saveProductsCSV(state);
                }
                else {
                    std::cout << "Product ID mapping missing in active warehouse slots." << std::endl;
                }
            }
        }
        else if (supChoice == 3) {
            std::cout << activeSup->generateInvoice() << std::endl;
        }
    }
}
